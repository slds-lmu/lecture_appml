\begin{frame}{Cost-Sensitive learning: In a Nutshell}
	%	
	\scriptsize{

		\begin{itemize}
		
    		\item Cost-sensitive learning: 
                \begin{itemize}
                    \scriptsize
                    \item Classical learning: data sets are balanced, and all errors have equal costs
                    \item We now assume given, unequal cost
                    \item And try to minimize them in expectation

                \end{itemize}
    		
    		\item Applications:
      
    		\begin{itemize}
    			\scriptsize	
    			\item Medicine --- Misdiagnosing as healthy vs. having a disease
    			\item (Extreme) Weather prediction ---  Incorrectly predicting that no hurricane occurs 
    			\item Credit granting --- Lending to a risky client vs. not lending to a trustworthy client.
    		\end{itemize}
         
		
		\end{itemize}
        \vspace{15pt}

        \begin{minipage}{0.49\textwidth}
            \begin{table}[]
                \centering
                \begin{tabular}{p{1cm}c|cc}
                    & &\multicolumn{2}{c}{Truth} \\
                    & & Default & Pays Back  \\
                    \hline
                    \multirow{2}{*}{\parbox{1cm}{Pred.}} & Default & 0 & $ 10 $\\
                    & Pays Back & $1000$ & $0$   \\
                \end{tabular}
            \end{table}
        \end{minipage}
        \hfill
        \begin{minipage}{0.49\textwidth}
            \begin{itemize}

                \item In these examples, \textbf{the costs of a false negative is much higher than the costs of a false positive}.
                \vspace{15pt}
                
                \item In some applications, the costs are \textbf{unknown} $\leadsto$ need to be specified by experts, or be learnt.
            \end{itemize}

\begin{frame}{Cost matrix}
%	
%	
	\begin{itemize}
%		
		\item Input: cost matrix $\mathbf{C}$ 
	\end{itemize}
	%	
	\begin{center}
		\tiny
		\begin{tabular}{cc|>{\centering\arraybackslash}p{8em}>{\centering\arraybackslash}p{8em}>{\centering\arraybackslash}p{5em}>{\centering\arraybackslash}p{8em}}
			& & \multicolumn{4}{c}{\bfseries True Class $y$} \\
			&  & $1$ & $2$ & $\ldots$ & $g$  \\
			\hline
			\bfseries Classification     & $1$ & $C(1,1)$  &  $C(1,2)$  & $\ldots$ &  $C(1,g)$ \\
			& $2$ &  $C(2,1)$  &  $C(2,2)$  & $\ldots$ & $C(2,g)$  \\
            $\yh$ & & & & & \\
			& $\vdots$ & $\vdots$ & $\vdots$ & $\ldots$ & $\vdots$ \\
			& $g$ & $C(g,1)$ & $C(g,2)$  & $\ldots$ &  $C(g,g)$\\
		\end{tabular}
	\end{center}
	%	
	\begin{itemize}
		%		
		\item $C(j,k)$ is the cost of classifying class $k$ as $j,$ 
  \item 0-1-loss would simply be: $C(j,k) = \mathds{1}_{[ j \neq k ]}$
		
		\item $\mathbf{C}$ designed by experts with domain knowledge
		\begin{enumerate}
            \item Too low costs: not enough change in model, still costly errors
            \item Too high costs: might never predict costly classes
		\end{enumerate}
		
	\end{itemize}

\begin{frame}{Cost matrix for Imbalanced Learning}
		\begin{itemize}			
	 
			\item Common heuristic for imbalanced data sets: 
	
			\begin{itemize}
                \item $C(j,k) = \frac{n_j}{n_k}$ with $n_k  \ll n_j$,\\
                misclassifying a minority class $k$ as a majority class $j$
		
				\item $C(j,k) = 1$ with $n_j \ll n_k$,\\
    misclassifying a majority class $k$ as a minority class $j$
			
				\item 0 for a correct classification 
			
			\end{itemize}
\vspace{1cm}
    		\item Imbalanced binary classification: \\
            \begin{table}[]
                \centering
                \begin{tabular}{cc|cc}
                    & &\multicolumn{2}{c}{True class} \\
                    & & $y=1$ & $y=-1$  \\
                    \hline
                    \multirow{2}{*}{\parbox{0.5cm}{Pred. class}}& $\hat y$ = 1     & $0$                & $ 1 $\\
                    & $\hat y$ = -1 & $ \frac{n_-}{n_+} $              &  $0$   \\
                \end{tabular}
            \end{table}
    		

            \item So: much higher costs for FNs

            \item \footnotesize \emph{Caveat:} the $n_j/n_k$ heuristic is one ad-hoc choice; tuning the decision threshold (Empirical Thresholding, later) is often a more principled alternative.
        \end{itemize}
		
\end{frame}

\begin{frame}{MECP: the intuition}
    \begin{itemize}
        \item \textbf{Standard ML:} predict the most \emph{probable} class.
        \item \textbf{Cost-sensitive ML} (MECP, \emph{Minimum Expected Cost Principle}): predict the \emph{cheapest action} -- the one with the lowest \textbf{expected cost}, averaging over the true class.

        \vspace{4pt}
        \item \emph{Concrete:} $\P(\text{benign})=0.7$, $\P(\text{malignant})=0.3$; missing cancer costs $100$, a false alarm costs $1$.
        \begin{itemize}
            \item $\E[\,\text{cost} \mid \text{predict benign}\,] = 0.7 \cdot 0 + 0.3 \cdot 100 = \mathbf{30}$
            \item $\E[\,\text{cost} \mid \text{predict malignant}\,] = 0.7 \cdot 1 + 0.3 \cdot 0 = \mathbf{0.7}$
        \end{itemize}

        \vspace{4pt}
        \item \textbf{Standard ML picks benign} (more probable). \textbf{MECP picks malignant} (vastly cheaper in expectation).
    \end{itemize}
\end{frame}


\begin{frame}{Minimum expected Cost Principle}

			\item Suppose we have:

                \item a cost matrix $\mathbf{C}$
                
                \item knowledge of the true posterior $p(\cdot ~|~ \xv)$

			\item Predict class \textcolor{blue}{$j$} with smallest expected costs when marginalizing over true classes \textcolor{orange}{$k$}:

		\end{itemize}

		$$  \hx := \argminlim_{\textcolor{blue}{j}=1,\ldots,g} \sumkg 	\pikx \, C(\textcolor{blue}{j},\textcolor{orange}{k}). $$
 
%        \item For $\xv$, making prediction $i$ means \textbf{acting as if $i$ is the true class of $\xv$.}

\begin{frame}{Optimal Threshold for Binary Case 1/2}
    \begin{itemize}
            \item Optimal decisions do not change if 
            \begin{itemize}
                \item $\mathbf{C}$ is multiplied by positive constant
                \item $\mathbf{C}$ is added with constant shift
            \end{itemize}

            \item Scale and shift $\mathbf{C}$ to get simpler $\mathbf{C}^\prime$: 
            \begin{table}[]
                \centering
                    \begin{tabular}{cc|cc}
        			& &\multicolumn{2}{c}{True class} \\
        			& & $y=1$ & $y=-1$  \\
        			\hline
        			\multirow{2}{*}{\parbox{0.5cm}{Pred.  class}} & $\hat y$ = 1 & $C^\prime(1,1)$ & $1$ \\
        			& $\hat y$ = -1 & $C^\prime(-1, 1)$ & 0\\
                \end{tabular}
            \end{table}
            where 
            \begin{itemize}
                \item $C^\prime (-1, 1) = \frac{C(-1, 1) - C(-1, -1)}{C(1, -1) - C(-1, -1)}$
                \item $C^\prime (1, 1) = \frac{C(1, 1) - C(-1, -1)}{C(1, -1) - C(-1,-1)}$
            \end{itemize}
    \end{itemize}
\end{frame}
\begin{frame}{Optimal Threshold for Binary Case 2/2}
            \begin{itemize}
            \item We predict $\xv$ as class $1$ if 
            \begin{align*}
                \E_{K \sim p(\cdot ~|~ \xv)}( C^\prime(1,K) )  \leq \E_{K \sim p(\cdot ~|~ \xv)}( C^\prime(-1,K) ) \\
            \end{align*}

			\item Let's unroll the expected value and use $\mathbf{C}^\prime$:
            % need this footnotsize operation, otherwise the first line will be very long.
            \footnotesize{
    			\begin{align*}	
                    & p(-1 ~|~ \xv ) C^\prime(1,-1)  + 	p(1 ~|~ \xv ) C^\prime(1,1) \leq  p(-1 ~|~ \xv ) C^\prime(-1,-1)  + 	p(1 ~|~ \xv ) C^\prime(-1,1)  \\ 
                    &\Rightarrow [1 - p(1 ~|~ \xv)] \cdot 1 + p(1 ~|~ \xv) C^\prime(1, 1) \leq p(1 ~|~ \xv) C^\prime(-1, 1) \\
                    &\Rightarrow p(1 ~|~ \xv) \geq \frac{1}{C^\prime(-1, 1) - C^\prime(1, 1) + 1} \\
                    &\Rightarrow p(1 ~|~ \xv) \geq \frac{C(1, -1) - C(-1, -1)}{C(-1, 1) - C(1, 1) + C(1, -1) - C(-1, -1)} = c^*
    			\end{align*}
            }
		
            \item If even $C(1, 1) = C(-1, -1) = 0 $, we get:
            \begin{align*}
                p(1 ~|~ \xv) \geq \textcolor{blue}{\frac{C(1, -1)}{C(-1, 1) + C(1, -1)}} = \textcolor{blue}{c^{*}}
            \end{align*}

            \item Optimal threshold \textcolor{blue}{$c^*$} for probabilistic classifier

            $$   \hx := 2 \cdot \mathds{1}_{[ \pi(\xv) \geq \textcolor{blue}{c^*}]} -1 $$
							
		\end{itemize}
\end{frame}

\begin{frame}{Optimal threshold $c^*$: the intuition}
    \small
    \begin{itemize}
        \item The derivation above gives a clean threshold:
        $$c^* = \frac{C(\text{FP})}{C(\text{FP}) + C(\text{FN})}.$$

        \item Plain English: \textbf{the costlier FNs are relative to FPs, the lower the threshold} -- ``lower the bar'' for predicting positive.
    \end{itemize}

    \begin{figure}
        \centering
        \includegraphics[width=0.55\linewidth]{figure/threshold_cstar_intuition.pdf}
    \end{figure}

    \footnotesize Equal costs $\Rightarrow c^* = 0.5$ (the standard threshold). FN $10\times$ FP $\Rightarrow c^* \approx 0.09$ (predict positive even at $9\%$ confidence).
\end{frame}


\begin{frame}{Empirical Thresholding: binary case}
    \begin{itemize}
        \item Theoretical threshold from MECP not always best, due to e.g. wrong model class, finite data, etc.
        \item Simply measure costs on data with different thresholds
        \item Then pick best threshold \furtherreading{SHENG2006} ($T$ = threshold, $M_c$ = empirical misclassif.\ cost):
        \begin{figure}[h]
            \centering
            \includegraphics[width=0.95\textwidth]{figure/threshold_adjusting.png}
        \end{figure}

        %\item The optimal threshold $T^{*}$ corresponds to the point with the smallest $M_C$.

        \item What if two equal local minima? Prefer the one with wider span -- less sensitive to small threshold changes (more robust in $T$).

        \item Do this on validation data / over cross-val to avoid overfitting!
    \end{itemize}
\end{frame}

\begin{frame}{Empirical Thresholding: Multiclass}
    \begin{itemize}
 %        \item It is also possible to perform threshold adjusting in multi-class classifcation.
        
  %      \item Here, we present a simple approach for classifiers that output scores $\pi(\xv) = (\pi(\xv)_1,\ldots,\pi(\xv)_g)^\top$ in multi-class classification:
        
            \item In the standard setting, we predict class $h(\xv) = \argmaxlim_{k} \pikx$.
            
            \item Let's use $g$ thresholds $c_k$ now 
            
            \item Re-scale scores $ \mathbf{s} = (\frac{\pi(\xv)_1}{c_1},\ldots, \frac{\pi(\xv)_g}{ c_g})^\top$, 
            \item Predict class $\argmaxlim_k s_k = \argmaxlim_k \frac{\pikx}{c_k}$.

            \item Compute empirical costs over cross-validation

            \item Optimize over $g$ (actually: $g-1$) dimensional threshold vector $(c_1, \ldots, c_g)^T$ to produce minimal costs

    \end{itemize}
\end{frame}

\begin{frame}{Binary Instance-specific cost Learning}
    \begin{itemize}
        \item Assumes instance-specific costs for every observation:  $\D^{(n)} = \{(\xi, \cv^{(i)})\}_{i=1}^n$, where $(\xi, \cv^{(i)}) \in \mathbb{R}^p \times \mathbb{R}^2$.

        \item Define ``true class'' as cost minimal class
        
        \item Define observation weights: $|\cv^{(i)}[1] - \cv^{(i)}[0]|$
        % Consider four instances, where $\cv^{(i)}[0]$ and $\cv^{(i)}[1]$ denote the instance-wise cost of classifying an example as 0 or 1 respectively, $\yi$ the true label and 

        \begin{center}
            \begin{tabular}{cc|cccc}\
        			& & $\cv^{(i)}[0]$ & $\cv^{(i)}[1]$ & $\yi$ & $w^{(i)}$ \\
        			\hline & $\xv^{(1)}$ & 1 & 1 & 0 & 0\\
        			& $\xv^{(2)}$ & 1 & 2 & 0 & 1\\
        			& $\xv^{(3)}$ & 7 & 3 & 1 & 4\\
            \end{tabular}
        \end{center}

        \item Now solve weighted ERM:
        \begin{equation*}
            \mathcal{R}_{emp}(\boldsymbol{\theta}) = \sum_{i=1}^{n}w^{(i)} \Lxyit
        \end{equation*}
        
        \item NB: Instances with equal costs are effectively ignored.
        \end{itemize}
            
\end{frame}

\begin{frame}{CSOVO: the big picture}
    \begin{itemize}
        \item Cost-sensitive \textbf{binary} classification is solved -- it reduces cleanly to \emph{weighted} binary classification.

        \item Cost-sensitive \textbf{multiclass} is much harder: arbitrary cost matrix, no clean general method.

        \item \textbf{CSOVO's strategy:} reduce the multiclass problem to many small binary ones, then aggregate.
    \end{itemize}

    \vspace{12pt}
    \begin{center}
        \fbox{\parbox{0.21\linewidth}{\centering multiclass\\cost data}}
        \;$\Longrightarrow$\;
        \fbox{\parbox{0.27\linewidth}{\centering $\frac{g(g-1)}{2}$ weighted\\binary problems}}
        \;$\Longrightarrow$\;
        \fbox{\parbox{0.17\linewidth}{\centering majority\\vote}}
    \end{center}
    \vspace{12pt}

    \begin{itemize}
        \item \textbf{Why it works:} a multiclass error forces many pairwise errors $\Rightarrow$ accurate pairwise classifiers give an accurate multiclass classifier (formal bound shown later).
    \end{itemize}
\end{frame}


\begin{frame}{CSOVO: what each pairwise classifier does}
    \small
    For each class pair $(j, k)$, train one binary classifier $f^{(j,k)}$.

    \vspace{10pt}
    \textbf{The question it answers:}
    \begin{center}
        \emph{``For this instance, is class $j$ cheaper or class $k$ cheaper?''}\\[3pt]
        (\textbf{not} ``which is the true class?'' -- the key conceptual shift)
    \end{center}

    \vspace{8pt}
    \textbf{How each instance contributes:}
    \begin{itemize}
        \item \textbf{Pseudo-label} = whichever of $\{j, k\}$ is \textcolor{green}{cheaper} for this instance.
        \item \textbf{Weight} = cost gap $|\cv^{(i)}[j] - \cv^{(i)}[k]|$.
        \item \textbf{Equal costs} $\Rightarrow$ weight $0$ $\Rightarrow$ \textcolor{gray}{dropped} (no preference, no signal).
    \end{itemize}

    \vspace{8pt}
    \textbf{At prediction time:} all $\frac{g(g-1)}{2}$ classifiers vote, majority wins.
\end{frame}

\begin{frame}{CSOVO}
    \begin{itemize}
        \item When training a binary classifier $f^{(j, k)}$ for class $j$ vs. $k$, turn each instance into a \textbf{weighted binary example}:
        \begin{itemize}
            \item Pseudo-label $\tilde{y}^{(i)} = \argmin_{l \in \{j, k\}} \cv^{(i)}[l]$: the \emph{cheaper} of the two classes for this instance (\textbf{not} its true label!).

            \item Sample weight $w^{(i)} = |\cv^{(i)}[j] - \cv^{(i)}[k]|$: the cost gap between the two options.

            \item \emph{Color convention:} \textcolor{green}{cheaper}, \textcolor{red}{pricier}, \textcolor{gray}{dropped (equal costs)}.

\begin{frame}{CSOVO}
    \begin{itemize}
    \item Example continued (pair $1$ vs $3$):
    \begin{center}  
        \footnotesize{
            \begin{tabular}{cc|ccc|ccc}
                & & $\cv^{(i)}[1]$ & $\cv^{(i)}[2]$ & $\cv^{(i)}[3]$ & $(\cv^{(i)}[1], \cv^{(i)}[3])$ & $\tilde{y}^{(i)}$ & $w^{(i)}$\\
                \hline & $\xv^{(1)}$ & 0 & 2 & 3 & (\textcolor{green}{0}, \textcolor{red}{3}) & 1 & 3\\
                & $\xv^{(2)}$ & 1 & 0 & 1 & \textcolor{gray}{(1, 1)} & \textcolor{gray}{--} & \textcolor{gray}{0} \\
                & $\xv^{(3)}$ & 2 & 0 & 3 & (\textcolor{green}{2}, \textcolor{red}{3}) & 1 & 1\\
            \end{tabular}
        }
    \end{center}

    \item $\xv^{(2)}$ has equal costs $(1, 1) \Rightarrow$ weight $0 \Rightarrow$ dropped for this pair (no preference between $1$ and $3$).

    \item Wrap everything up:
    \begin{enumerate}
        \item For class $j$ vs. $k$, transform all $(\xi, \cv^{(i)})$ to $(\xi, \argmin_{l \in \{j, k\}} \cv^{(i)}[l])$ with sample-wise weight $|\cv^{(i)}[j] - \cv^{(i)}[k]|$.
        
        \item Train a weighted binary classifier $f^{(j, k)}$ using the above
        
        \item Repeat step 1 and 2 for different $(j, k)$.
        
        \item Predict using the votes from all $f^{(j, k)}$.
    \end{enumerate}

    \item Theoretical guarantee: total test cost $\leq 2\sum_{j < k}$ (test cost of $f^{(j, k)}$).
    \item[] {\footnotesize Intuition: a multiclass error costs at most (twice) the sum of the pairwise errors that caused it $\leadsto$ good pairwise classifiers give a good multiclass one.}
    \end{itemize}
\end{frame}


\begin{frame}{CSOVO: closing the example \& when to use it}
    \small
    \textbf{Closing the example.} If each pairwise classifier predicts the cheaper of its two classes, the votes for the three training instances roll up to:

    \begin{center}
        \footnotesize
        \begin{tabular}{c|c|ccc|c}
             & $\cv^{(i)}$ & $f^{(1,2)}$ & $f^{(1,3)}$ & $f^{(2,3)}$ & majority winner \\
            \hline
            $\xv^{(1)}$ & $(\textcolor{green}{\textbf{0}}, 2, 3)$ & 1 & 1 & 2 & \textbf{\textcolor{green}{1}} \\
            $\xv^{(2)}$ & $(1, \textcolor{green}{\textbf{0}}, 1)$ & 2 & either$^*$ & 2 & \textbf{\textcolor{green}{2}} \\
            $\xv^{(3)}$ & $(2, \textcolor{green}{\textbf{0}}, 3)$ & 2 & 1 & 2 & \textbf{\textcolor{green}{2}} \\
        \end{tabular}
    \end{center}
    {\footnotesize $^*$ for $\xv^{(2)}$, pair $1$ vs $3$ has tied costs $(1, 1)$; the classifier was trained without this instance and either vote leaves the winner unchanged.}

    \begin{itemize}
        \item All three winners match the cheapest class.
        \item \textbf{When CSOVO helps:} costs matter \emph{and} the base learner makes meaningful errors. CSOVO beats vanilla OVO on 7/10 benchmark datasets \furtherreading{LIN2014-CSOVO}.
        \item \textbf{When it can hurt:} if the base learner is already near-perfect, the pseudo-labeling introduces noise that outweighs the cost-information gain. \emph{Don't reach for CSOVO if vanilla OVO already solves the problem.}
    \end{itemize}
\end{frame}
